class Session(_CachedSession):

    def __init__(self,
        name: str,
        max_age: int = -1,
        adapter: Adapter = None
    ) -> None:
        from ..pc._pc import loc

        super().__init__(
            cache_name = loc.cache.child(f'{name}.sqlite').path,
            backend = 'sqlite',
            expire_after = max_age
        )

        if adapter:
            self.mount("http://", adapter)
            self.mount("https://", adapter)
