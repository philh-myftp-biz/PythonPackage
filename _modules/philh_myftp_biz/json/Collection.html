    @contextmanager
    def handle(self) -> Generator[STRUCT, None, None]:
        data = self.read()
        try:

        finally:
            self.save(data)

    def save(self, data: 'STRUCT | Collection[T, STRUCT]'=_NO_VALUE) -> None:

        if data is _NO_VALUE:
            data = self.read()

        if isinstance(data, Collection):
            data = data.read()

        self._cache = data

        if hasattr(self, 'var'):
            self.var.save(data)
