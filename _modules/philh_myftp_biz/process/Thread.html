class Thread[T]:

    result: T

    @property
    def _Builder(self):
        from threading import Thread
        return Thread

    def __init__(self,
        func: Callable[..., T],
        *args: str,
        **kwargs: str
    ) -> None: 

        self._Builder.run = lambda s: setattr(
            self, 
            'result', 
            s._target(*s._args, **s._kwargs)
        )

        # Create new thread
        self.p = self._Builder(
            target = func,
            kwargs = kwargs,
            args = args
        )

        # Close when main execution ends
        self.p.daemon = True

        # start the task
        self.p.start()

        self.wait = self.p.join

        self.running: bool = property(self.p.is_alive)

class Watcher[T](Thread):

    def __init__(self,
        checker: Callable[[], T],
        handler: Callable[[T], None],
        interval: float = .3
    ) -> None:
        
        self.checker = checker
        self.handler = handler
        self.interval = interval

        super().__init__(self._main)

    def _main(self) -> None:
        from inspect import signature
        from timelib2 import sleep

        lvalue: T = None

        try:
            params = signature(self.handler).parameters
        except TypeError:
            params = []
        
        while sleep(self.interval):

            value = self.checker()

            if value != lvalue:

                lvalue = value

                if len(params) == 0:
                    self.handler()
                else:
                    self.handler(value)

def Alive() -> bool:
    """Check if the main thread is running"""
    from threading import main_thread

    return main_thread().is_alive()
